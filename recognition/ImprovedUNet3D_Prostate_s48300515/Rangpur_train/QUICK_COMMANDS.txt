## ON RANGPUR (After SSH):

# Navigate to project:
cd ~/ImprovedUNet3D_Prostate

# Run setup (FIRST TIME ONLY):
bash setup_rangpur.sh

# Activate environment (every time you login):
conda activate prostate_seg

# Test your setup:
sbatch test_job.sh

# Start training:
sbatch train_job.sh

# Check job status:
squeue -u s4830051

# Monitor training progress:
tail -f train_*.out

# Cancel a job:
scancel JOBID

# After training, generate predictions:
sbatch predict_job.sh

# List all outputs:
ls -lh outputs_a100/

# View training results:
cat outputs_a100/test_results.json
